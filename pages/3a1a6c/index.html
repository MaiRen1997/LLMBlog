# 创建项目

  1. 新建文件夹
  2. 使用uv init 初始化项目
  3. 使用uv venv 或者uv venv --python 3.12 安装指定python版本的虚拟环境
  4. 使用uv sync 同步依赖
  5. 使用uv add添加项目依赖

# 使用FastAPI定义简单项目

# 使用简单的FastAPI

from fastapi import FastAPI

app = FastAPI(title="My Awesome API", version="0.1.0")

@app.get("/")
def read_root():
    return {"message": "Hello, World!"}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)

# 使用路由

  1. 使用APIRouter，及prefix、tags(值是list)相关属性定义接口

  2. 使用其实例，定义实际接口

    from fastapi import FastAPI, APIRouter

router = APIRouter(prefix='/test', tags = ['测试接口'])
@router.get("/responseData")
def responseData():
    return {
        'code': 200,
        'message': 'success',
        'data': {'order': 'TN001'}
    }

  3. 在main程序中，导入上面的router实例，并在fast 实例中引用

    from test import router
app = FastAPI(title='my API', version='1.0')

app.include_router(router)

# 请求第三方接口

  1. 使用request库请求接口
  2. response.raise_for_status()
    1. 如果状态码是 200-399，什么也不发生
    2. 如果状态码是 400 或以上，抛出 HTTPError 异常
    3. 使用except HTTPError as http_err:捕获异常
import requests

def get_baidu_data():
    """
    调用测试接口并返回数据。
    """
    print('调用了该测试方法')
    try:
        response = requests.get('http://localhost:5000/test/responseData', timeout=10)
        response.raise_for_status()
        return response.json()
    except HTTPError as http_err:
        print(f"HTTP 错误发生: {http_err}")
        print(f"状态码: {response.status_code}")
    except requests.RequestException as e:
        return {"error": f"Error fetching data: {str(e)}"}

# 引入变量

  1. 在根目录定义环境变量

    OLLAMA_BASE_URL = 'http://localhost:11434/v1'
OLLAMA_API_KEY = 'ollama'

  2. 在模型文件夹中使用该变量

    import os
import dotenv

from openai import OpenAI

# 加载环境变量
dotenv.load_dotenv()

# 使用环境变量
base_url = os.getenv('OLLAMA_BASE_URL', 'http://localhost:11434/v1')
api_key = os.getenv('OLLAMA_API_KEY', 'ollama')


client = OpenAI(
    base_url=base_url,
    api_key=api_key,
)
