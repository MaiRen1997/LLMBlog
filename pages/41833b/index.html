嵌入式模型

# 基本使用

  • embed_query： 用来转换文本('单次转换')
  • embed_documents： 用来批量转换(‘批量转换’)
from langchain_communicaty.embeddings import DashScopeEmbeddings

#初始化嵌入模型对象，其默认使用模型是：text-embedding-v1
embed = DashScopeEmbeddings(
	model="模型名称" # 如果没有该项，则使用text-embedding-v1
)

#测试
print(embed.embed_query('请将这段文本翻译成中文'))
print(embed.embed_documents(['文本1', '文本2', '文本3']))
