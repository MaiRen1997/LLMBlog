# main方法

  1. 使用uuid.uuid4()，创建对话用的thread_id，并传入到agent实例
  2. 使用while 死循环，进行多轮对话
  3. 通过判断用户输入，是否包含quit，来决定是否需要结束对话
    """
    AplantAgent 主程序入口
    """
    import uuid

    from agent.agent import run_agent

    def main():
        session_id = str(uuid.uuid4())
        """主函数"""
        while True:
            print("=" * 60)
            print("开启新一轮会话，需要终止，请键入quit")
            print("=" * 60)
            user_input = input(f"对话ID:{session_id},请告知我您想要了解什么问题:")
            if('quit' in user_input):
                break
            run_agent(user_input, thread_id=session_id)

    if __name__ == "__main__":
        main()

"""
使用LangGraph框架的AI Agent实例
集成DeepSeek API，支持工具调用
使用 create_react_agent 简化架构
"""

import os
from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain.agents import create_agent
from langgraph.checkpoint.memory import InMemorySaver

# 加载 .env 文件
load_dotenv()

# 从 .env 文件读取配置
DEEPSEEK_API_KEY = os.getenv("DEEPSEEK_API_KEY")
DEEPSEEK_API_BASE_URL = os.getenv("DEEPSEEK_API_BASE_URL")

# ============= 定义工具 =============
from tools.sendEmail import send_email
from tools.searchInformation import search_information
from tools.getCurrentTime import get_current_time

# 工具列表
tools = [send_email, get_current_time, search_information]

# ============= 初始化LLM =============
def create_llm():
    """创建LLM实例，使用DeepSeek API"""
    return ChatOpenAI(
        api_key=DEEPSEEK_API_KEY,
        base_url=DEEPSEEK_API_BASE_URL,
        model="deepseek-chat",
        temperature=0.7,
    )

# ============= 创建Agent =============
# 全局 checkpointer 实例，确保跨请求内存持久化
_checkpointer = InMemorySaver()

def create_my_agent():
    """使用 create_agent 创建 Agent"""
    llm = create_llm()
    # create_agent 会自动处理 AgentState (messages)
    # 并集成工具调用和循环逻辑
    return create_agent(llm, tools, checkpointer=_checkpointer)

# 创建全局 Agent 实例
_agent_instance = create_my_agent()

# ============= 主函数 =============
def run_agent(user_input: str, thread_id: str = "default_thread"):
    """运行Agent"""
    config = {"configurable": {"thread_id": thread_id}}
    
    # 构造输入消息
    input_messages = {"messages": [("user", user_input)]}

    print(f"\n用户输入: {user_input}")
    print("-" * 50)

    # 执行Agent
    # invoke 会根据 thread_id 自动从 checkpointer 加载历史记录
    result = _agent_instance.invoke(input_messages, config)

    # 获取最后一条消息（Agent的响应）
    final_response = result["messages"][-1]

    print(f"\nAgent响应: {final_response.content}")
    print("-" * 50)

    return result

# 定义方法

  1. 使用@tool装饰器装饰函数，将函数变为agent可使用的方法

  2. 定义方法时，需要在最上面，依次定义如下参数

    1. 描述：用于说明，该方法的作用
    2. 参数Args，具体说明每个参数的作用
    3. Returns：说明该函数的返回值
  3. from langchain_core.tools import tool
@tool
def send_email(recipient: str, subject: str, body: str) -> str:
    """
    发送电子邮件的工具

    Args:
        recipient: 收件人邮箱地址
        subject: 邮件主题
        body: 邮件正文

    Returns:
        邮件发送状态
    """
    print(f"发送邮件到: {recipient}")
    print(f"主题: {subject}")
    print(f"内容: {body}")
    return f"邮件已发送给 {recipient}"
