    界面操作

    # 元素通用操作

    # 获取文本内容

    # 获取元素可见文本
page.locator('#source').inner_text()

# 获取多个元素的可见文本列表
page.locator('.item').all_inner_texts()

# 获取元素内所有文本（包括隐藏文本
page.locator('#source').text_content()

# 获取多个元素的所有文本
page.locator('.item').all_text_contents()

    # 获取元素属性

    page.locator('a').get_attribute('href')

    # 获取元素内部 HTML

    page.locator('#source').inner_html()

    # 点击 / 双击 / 悬停

    # 单击
page.locator('#btn').click()

# 双击
page.locator('#btn').dblclick()

# 悬停
page.locator('#btn').hover()

    # 等待元素可见 / 判断元素可见

    # 等待元素可见（默认 state='visible'，超时 30 秒
page.locator('#source').wait_for()

# 等待元素消失
page.locator('#source').wait_for(state='hidden')

# 指定超时时间
page.locator('#source').wait_for(timeout=5000)

# 立即判断是否可见（不等待）
if page.locator('#source').is_visible():
    print('存在')

    # 输入框操作

    # 单行 / 多行文本输入框
page.locator('input[type=text]').fill('hello')

# 清空输入框
page.locator('input[type=text]').clear()

# 获取输入框里的文字
text = page.locator('input[type=text]').input_value()

    # 文件输入框

    # 单选文件
page.locator('input[type=file]').set_input_files('d:/1.png')

# 多选文件
page.locator('input[type=file]').set_input_files(['d:/1.png', 'd:/2.jpg'])

    # radio 单选 / checkbox 多选

    # 点选 radio
page.locator('input[type=radio][value=小雷老师]').check()

# 判断是否选中
page.locator('input[type=radio][value=小雷老师]').is_checked()

# 取消选择（checkbox 适用）
page.locator('input[type=checkbox][value=小江老师]').uncheck()

# 获取当前选中的选项
locs = page.locator('#s_radio input[name=teacher]:checked').all()
teachers = [lc.get_attribute('value') for lc in locs]
print('当前选中的是:', ' '.join(teachers))

    # select 元素操作

    # 单选：直接选择文本或 value
page.locator('#ss_single').select_option('小江老师')

# 根据 value
page.locator('#ss_single').select_option(value='小江老师')

# 根据 label（选项文本）
page.locator('#ss_single').select_option(label='小江老师')

# 根据索引
page.locator('#ss_single').select_option(index=1)

# 多选：传入列表
page.locator('#ss_multi').select_option(['小江老师', '小雷老师'])
page.locator('#ss_multi').select_option(value=['小江老师', '小雷老师'])
page.locator('#ss_multi').select_option(label=['小江老师', '小雷老师'])

# 清空所有选择
page.locator('#ss_multi').select_option([])

# 获取所有选中的选项
page.locator('#ss_multi').select_option(['小江老师', '小雷老师'])
texts = page.locator('#ss_multi option:checked').all_inner_texts()
print(texts)

    # 网页操作

    # 打开网址 / 刷新 / 后退 / 前进
page.goto('https://www.byhy.net')
page.reload()
page.go_back()
page.go_forward()

# 获取网页 HTML
html = page.content()

# 获取页面标题
title = page.title()

# 设置浏览器窗口大小（像素）
page.set_viewport_size({'width': 1280, 'height': 720})
