LangSmith

# 作用

  1. 能够检查链或者Agent内部情况的变更

# 为什么使用

  1. 因为agent封装了整个调用工具的交互流程，不好看到调用agent整体和llm的交互有哪些，通过langSmith就可以清晰的看到

# 使用

# 创建Apikey

在langSmith申请Apikey (opens new window)

# 安装依赖

pip install --pre -U langchain langchain-openai

# 配置环境变量

  1. LANGSMITH_PROJECT：追踪数据的存放位置
export LANGSMITH_TRACING=true
export LANGSMITH_ENDPOINT=https://api.smith.langchain.com
export LANGSMITH_API_KEY=<your-api-key>
export LANGSMITH_PROJECT="pr-downright-questionnaire-83"
export OPENAI_API_KEY=<your-openai-api-key>

# 运行agent

  1. 配置环境变量后，重新load
  2. 之后langchain底层就可以自动将每次调用，记录到langSmith当中了
from langchain.agents import create_agent


def get_weather(city: str) -> str:
    """Get weather for a given city."""
    return f"It's always sunny in {city}!"


agent = create_agent(
    model="openai:gpt-5-mini",
    tools=[get_weather],
    system_prompt="You are a helpful assistant",
)

# Run the agent
agent.invoke(
    {"messages": [{"role": "user", "content": "What is the weather in San Francisco?"}]}
)
