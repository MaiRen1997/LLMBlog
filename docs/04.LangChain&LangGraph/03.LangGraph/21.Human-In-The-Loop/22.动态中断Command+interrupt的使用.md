## 过程概述

1. 定义一个节点函数
   1. 该节点函数含有`interrupt`关键字，如`user_response = interrupt("请批准这个操作 (yes/no): ")`
   2. 将获取到的用户输入，进行修改State或者其他计算逻辑中
2. 创建图实例
   1. 通过StateGraph创建图
   2. 添加节点
   3. 添加边
   4. 图编译
3. 通过Http请求/Input阻塞，获取用户输入
4. 将用户输入，通过`graph.invoke`，传入图
   1. 第一个参数是`Command`包装的输入
   2. 第二个参数是检查恢复点的id
   3. `result = graph.invoke(Command(resume=user_decision, checkpoint_id=checkpoint_id))`

## 定义节点(含有interrupt)

```python
from typing import TypedDict, List, Optional
import uvicorn
from langgraph.types import interrupt
from langgraph.graph import StateGraph, START, END
from langgraph.types import Command
from langgraph.checkpoint.memory import InMemorySaver
from fastapi import FastAPI,HTTPException
import uuid
thread_id = str(uuid.uuid4())
result = {}
app = FastAPI()
class State(TypedDict):
    messages: List[str]
    approved: bool
    step: int
    user_input: Optional[str]
    status: str,
    otherMsg: ''


# 定义节点1：需要用户批准的节点
def approval_required_node(state: State) -> dict:
    """这个节点会触发中断，等待用户批准"""
    print(f"\n[审批节点] 当前步骤: {state['step']}")
    print(f"[审批节点] 消息: {state['messages'][-1] if state['messages'] else '无消息'}")

    # 使用 interrupt 中断执行，等待用户输入
    # 注意：interrupt 必须在图节点内部调用
    user_response = interrupt("请批准这个操作 (yes/no): ")

    # 根据用户输入更新状态
    if user_response.lower() == "yes":
        state["approved"] = True
        state["messages"].append("✅ 操作已批准")
        print("[审批节点] 操作已批准")
    else:
        state["approved"] = False
        state["messages"].append("❌ 操作被拒绝")
        print("[审批节点] 操作被拒绝")

    state["step"] += 1
    state["user_input"] = user_response
    return state
```

## 构建图

```python
def create_graph():
    # 创建状态图
    builder = StateGraph(State)
    # 添加节点
    builder.add_node("approval", approval_required_node)
    # 添加边
    builder.add_edge(START, "approval")
    builder.add_edge("approval", END)

    # 使用内存检查点来支持中断和恢复
    memory = InMemorySaver()

    # 编译图，指定中断点
    # graph = builder.compile(checkpointer=memory, interrupt_before=["approval"])
    graph = builder.compile(checkpointer=memory)

    return graph
```

### 通过Http接口获取参数

```python
graph = create_graph()
@app.post("/chat")
def chat(request: dict):
    """启动对话流程"""
    config = {"configurable": {"thread_id": thread_id}}

    init_data = {
        "messages": [request.get("input", "")],
        "approved": False,
        "step": 0,
        "user_input": request.get("input", ""),
        "status": "pending"
    }
    result = graph.invoke(
        init_data,
        config=config
    )

    snapshot = graph.get_state(config)

    print(snapshot)
    if snapshot.interrupts:
        return {
            "need_approval": True,
            "thread_id": config["configurable"]["thread_id"]
        }
    print("✅ 图执行完成，没有中断")
    return {
        "completed": True,
        "result": result
    }
from pydantic import BaseModel
class ResumeRequest(BaseModel):
    thread_id: str
    user_decision: str
# 用户在前端点击"批准"后
@app.post("/resume")
def resume(resume_request: ResumeRequest):
    """恢复中断的流程"""
    config = {
        "configurable": {
            "thread_id": resume_request.thread_id,
        }
    }

    try:
        # 使用 Command.resume 恢复执行
        result = graph.invoke(
            Command(resume=resume_request.user_decision),
            config=config
        )
        return {
            "completed": True,
            "approved": result.get('approved'),
            "messages": result.get('messages'),
            "final_state": result
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
