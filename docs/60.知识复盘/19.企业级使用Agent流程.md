## 业务流程

```sh
前端
  ↓
FastAPI
  ↓
Session Manager
  ↓
Task Manager
  ↓
LangGraph
  ↓
LLM Stream
```

关键：

```
每个用户请求
都对应一个 asyncio task
```

用户点击停止：

```
cancel task
```

## 项目结构

```sh
project/
│
├── app.py
├── graph.py
├── task_manager.py
└── requirements.txt
```

## 代码实现

### Graph.py

```python
from typing import TypedDict
from langgraph.graph import StateGraph, END
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(
    model="gpt-4.1-mini",
    streaming=True
)

class AgentState(TypedDict):
    messages: list

async def chatbot_node(state: AgentState):

    response = await llm.ainvoke(state["messages"])

    return {
        "messages": state["messages"] + [response]
    }

builder = StateGraph(AgentState)

builder.add_node("chatbot", chatbot_node)

builder.set_entry_point("chatbot")

builder.add_edge("chatbot", END)

graph = builder.compile()
```

### task_manager.py（核心）

1. 使用`task = asyncio.create_task`创建任务
2. `tasks.get(session_id)`获取任务
3. `task.cancel()`取消任务
4. `tasks.pop(session_id, None)`移除任务

```python
import asyncio
from typing import Dict

class TaskManager:

    def __init__(self):

        # session_id -> task
        self.tasks: Dict[str, asyncio.Task] = {}

    def create_task(self, session_id: str, coro):

        # 如果已有旧任务
        self.cancel_task(session_id)

        task = asyncio.create_task(coro)

        self.tasks[session_id] = task

        return task

    def cancel_task(self, session_id: str):

        task = self.tasks.get(session_id)

        if task and not task.done():

            task.cancel()

            print(f"任务已取消: {session_id}")

    def remove_task(self, session_id: str):

        self.tasks.pop(session_id, None)

task_manager = TaskManager()
```

### app.py（重点）

1. 使用`task_manager`

   1. 使用`task_manager.create_task`创建任务

      ```python
      task = task_manager.create_task(
              session_id,
              stream_generator()
          )
      ```

   2. 使用`task_manager.cancel_task(session_id)`停止任务

```python
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
from graph import graph
from task_manager import task_manager

import asyncio
import json
import uuid

app = FastAPI()

# =========================
# 聊天接口
# =========================

@app.post("/chat")

async def chat(request: Request):

    body = await request.json()

    user_message = body["message"]

    session_id = body.get("session_id")

    if not session_id:
        session_id = str(uuid.uuid4())

    async def stream_generator():

        try:

            async for chunk in graph.astream(
                {
                    "messages": [
                        ("user", user_message)
                    ]
                },
                stream_mode="messages"
            ):

                # 检查客户端是否断开
                if await request.is_disconnected():

                    print("客户端已断开")

                    raise asyncio.CancelledError()

                yield f"data: {json.dumps(str(chunk))}\n\n"

        except asyncio.CancelledError:

            print(f"任务被取消: {session_id}")

            raise

        finally:

            task_manager.remove_task(session_id)

            print(f"任务清理完成: {session_id}")

    task = task_manager.create_task(
        session_id,
        stream_generator()
    )

    return StreamingResponse(
        task,
        media_type="text/event-stream"
    )

# =========================
# 停止接口
# =========================

@app.post("/stop/{session_id}")

async def stop_chat(session_id: str):

    task_manager.cancel_task(session_id)

    return {
        "success": True,
        "message": f"{session_id} 已停止"
    }
```
