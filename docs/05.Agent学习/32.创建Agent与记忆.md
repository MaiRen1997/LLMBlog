## Agent使用

### 创建Agent

1. TavilySearch是搜索工具，可以使用`max_results`返回最大搜索条目

```python
# pip install langchain-tavily
import os
from langchain_tavily import TavilySearch
from langchain.agents import create_agent
from langchain.chat_models import init_chat_model

# 定义模型
llm = init_chat_model(
    model="z-ai/glm-4.5-air:free",
    model_provider="openai",
    base_url="https://openrouter.ai/api/v1",
    api_key=os.getenv("OPENROUTER_API_KEY"),
)

# 定义 Tavily 搜索工具
search = TavilySearch(max_results=5)
tools = [search]

# 创建 Agent
agent = create_agent(
    model=llm,  # 模型
    tools=tools,  # 工具
    system_prompt="你是位助手，需要调用工具来帮助用户。",  # 系统提示词
)

```

### 使用Agent

```python
# 调用 Agent
res = agent.invoke(
    {"messages": [{"role": "user", "content": "今天北京的天气怎么样？"}]}
)
print(res)
```

## 添加记忆

### 说明

1. 在`创建agent`时，`传入checkpointer参数`，在`测试环境`下，可以`传入InMemorySaver()`，`生产环境`下，使用`数据库`来保存记忆
2. 在`invoke时`，需要指定，将记忆保存到哪个"会话"当中，`传入config参数：{"configurable": {"thread_id": "1"}}`，`thread_id`真正`指定`了将记忆`存入“哪个会话”`

### 使用内存

```python
import os
import datetime
from langchain_tavily import TavilySearch
from langchain.agents import create_agent
from langchain.chat_models import init_chat_model
from langgraph.checkpoint.memory import InMemorySaver

os.environ["LANGSMITH_PROJECT"] = "agent-with-memory"

# 定义 Tavily 搜索工具
search = TavilySearch(max_results=5)
tools = [search]

llm = init_chat_model(
    model="z-ai/glm-4.5-air:free",
    model_provider="openai",
    base_url="https://openrouter.ai/api/v1",
    api_key=os.getenv("OPENROUTER_API_KEY"),
)

# 创建 Agent
agent = create_agent(
    model=llm,
    tools=tools,
    checkpointer=InMemorySaver(),
)

# 调用
for chunk in agent.stream(
    input={
        "messages": [
            {
                "role": "system",
                "content": f"当前时间：{datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}",
            },
            {"role": "user", "content": "今天北京天气怎么样？"},
        ]
    },
    config={"configurable": {"thread_id": "1"}},
):
    print(chunk, end="\n\n")

for chunk in agent.stream(
    input={
        "messages": [
            {
                "role": "system",
                "content": f"当前时间：{datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}",
            },
            {"role": "user", "content": "上海呢？"},
        ]
    },
    config={"configurable": {"thread_id": "1"}},
):
    print(chunk, end="\n\n")
```

```python
# Redis 连接配置
REDIS_HOST = os.getenv("REDIS_HOST", "localhost")
REDIS_PORT = int(os.getenv("REDIS_PORT", 6379))
REDIS_DB = int(os.getenv("REDIS_DB", 0))
REDIS_PASSWORD = os.getenv("REDIS_PASSWORD", None)

# 创建 Redis 客户端
redis_client = redis.Redis(
    host=REDIS_HOST,
    port=REDIS_PORT,
    db=REDIS_DB,
    password=REDIS_PASSWORD,
    decode_responses=True  # 自动解码响应为字符串
)

# 定义生成 thread_id 的 hash 函数
def generate_thread_id(*args, **kwargs) -> str:
    """
    根据输入参数生成唯一的 thread_id
    可以基于用户ID、会话ID等信息生成
    """
    # 示例：基于用户ID和会话ID生成
    # 你可以根据需要自定义参数
    content = ""
    
    if args:
        content += "|".join(str(arg) for arg in args)
    if kwargs:
        content += "|" + "|".join(f"{k}={v}" for k, v in sorted(kwargs.items()))
    
    # 如果没有提供任何参数，使用时间戳
    if not content:
        content = str(datetime.datetime.now().timestamp())
    
    # 使用 SHA256 生成 hash
    thread_hash = hashlib.sha256(content.encode()).hexdigest()
    
    # 可以只取前16位或32位，便于查看
    return thread_hash[:32]

# 或者更简单的使用场景：基于用户会话
def generate_session_thread_id(user_id: str, session_id: str = None) -> str:
    """
    基于用户ID和会话ID生成 thread_id
    """
    if session_id is None:
        session_id = datetime.datetime.now().strftime("%Y%m%d")
    
    content = f"user:{user_id}:session:{session_id}"
    return hashlib.md5(content.encode()).hexdigest()[:16]

# 定义 Tavily 搜索工具
search = TavilySearch(max_results=5)
tools = [search]

# 初始化 LLM
llm = init_chat_model(
    model="z-ai/glm-4.5-air:free",
    model_provider="openai",

# 创建 Redis checkpoint saver
# 注意：需要安装 langgraph-checkpoint-redis
# pip install langgraph-checkpoint-redis

    redis_client=redis_client,
    # 可选：设置 checkpoint 的过期时间（秒）
    # ttl=3600 * 24 * 7,  # 7天过期
)

# 创建 Agent（使用 Redis 存储记忆）
agent = create_agent(
    model=llm,
    tools=tools,
    checkpointer=checkpointer,

# 示例1：使用简单的 hash 生成 thread_id
def demo_simple_hash():
    """基于简单字符串生成 thread_id"""
    # 例如：基于用户ID
    thread_id = generate_thread_id("user_123", "chat_session")
    print(f"生成的 thread_id: {thread_id}")
    
    print("\n=== 第一轮对话 ===")
    for chunk in agent.stream(
        input={
            "messages": [
                {
                    "role": "system",
                    "content": f"当前时间：{datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}",
                },
                {"role": "user", "content": "今天北京天气怎么样？"},
            ]
        },
        config={"configurable": {"thread_id": thread_id}},
    ):
        print(chunk, end="\n\n")
    
    print("\n=== 第二轮对话（相同 thread_id，应该有记忆）===")
    for chunk in agent.stream(

                {
                    "role": "system",

                {"role": "user", "content": "上海呢？"},
            ]
        },
        config={"configurable": {"thread_id": thread_id}},
    ):

    user_a_id = "user_alice"

    print(f"用户B的 thread_id: {session_b}")
    
    # 用户A对话

    for chunk in agent.stream(

            "messages": [

                {"role": "user", "content": "北京今天天气如何？"},

    # 用户B对话（不同的 thread_id，独立记忆）

# 示例3：自定义 thread_id 生成策略（推荐）
class ThreadIdManager:
    """Thread ID 管理器"""
    
    @staticmethod
    def from_user_session(user_id: str, session_id: str = None) -> str:
        """基于用户ID和会话ID"""
        if session_id is None:
            session_id = datetime.datetime.now().strftime("%Y%m%d")
        content = f"user:{user_id}:session:{session_id}"
        return hashlib.md5(content.encode()).hexdigest()
    
    @staticmethod
    def from_conversation(conversation_id: str) -> str:
        """基于对话ID"""
        return hashlib.sha256(f"conv:{conversation_id}".encode()).hexdigest()[:24]
    
    @staticmethod
    def from_request(request_id: str, user_agent: str = None) -> str:
        """基于请求ID和用户代理"""
        content = f"req:{request_id}"
        if user_agent:
            content += f":ua:{user_agent}"
        return hashlib.sha256(content.encode()).hexdigest()[:16]

def demo_advanced():
    """高级示例：使用 ThreadIdManager"""
    thread_mgr = ThreadIdManager()
    
    # 为当前用户创建 thread_id
    thread_id = thread_mgr.from_user_session("current_user", "chat_2025")
    print(f"使用管理器生成的 thread_id: {thread_id}")
    
    # 对话轮次
    questions = [
        "今天北京天气怎么样？",
        "那上海呢？",
        "对比一下这两个城市的气温"

                    {"role": "system", "content": f"当前时间：{datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"},
                    {"role": "user", "content": question},
                ]
            },
            config={"configurable": {"thread_id": thread_id}},
        ):
            print(chunk, end="\n\n")
```
